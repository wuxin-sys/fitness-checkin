### 测试打卡数据快速生成（Postman / 浏览器控制台）

在浏览器控制台（登录后）执行以下脚本可快速生成测试打卡：

```javascript
// 生成近30天随机打卡数据
const types = ['跑步', '胸部', '背部', '腿部', '肩部', '有氧', '其他']
const token = localStorage.getItem('token')

async function seedData() {
  for (let i = 0; i < 30; i++) {
    const date = new Date()
    date.setDate(date.getDate() - i)
    const dateStr = date.toISOString().split('T')[0]

    // 每天随机1-2种类型
    const count = Math.random() > 0.5 ? 2 : 1
    const shuffled = types.sort(() => Math.random() - 0.5)

    for (let j = 0; j < count; j++) {
      await fetch('/api/checkin', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': token
        },
        body: JSON.stringify({
          checkinDate: dateStr,
          sportType: shuffled[j],
          duration: Math.floor(Math.random() * 90) + 20,
          calorie: Math.floor(Math.random() * 500) + 100,
          remark: '测试数据'
        })
      })
    }
  }
  console.log('测试数据生成完毕')
}

seedData()
```
